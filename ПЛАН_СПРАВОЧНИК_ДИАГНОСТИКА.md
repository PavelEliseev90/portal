# План реализации справочника диагностики для интерактивной оценки

## Цель
Создать систему справочной информации для разделов диагностики, которая поможет сотрудникам:
- Правильно определять неисправности
- Понимать причины возникновения проблем
- Аргументировать клиентам необходимость ремонта
- Обучаться на примерах типичных случаев

## Текущая структура

### Модель данных
- `RegInteractivOcenkaTestDrive` - содержит поля диагностики:
  - `engine` (Двигатель)
  - `gear` (КПП)
  - `brake` (Тормозная система)
  - `suspension` (Подвеска)
  - `system_og` (Система выпуска ОГ)
  - `wheel` (Рулевое управление)
  - `conditioner` (Кондиционер)
  - `turbo` (Турбина)
  - `drive` (Полный привод)
  - `to` (ТО)
  - И соответствующие поля `comment_*` для комментариев

```python
class DiagnosticReferenceCategory(models.Model):
    """
    Категории справочника диагностики (Двигатель, КПП, Тормозная система и т.д.)
    """
    code = models.CharField('Код категории', max_length=50, unique=True)  # engine, gear, brake и т.д.
    name = models.CharField('Наименование', max_length=200)
    description = models.TextField('Описание', blank=True, null=True)
    order = models.IntegerField('Порядок сортировки', default=0)
    is_active = models.BooleanField('Активна', default=True)
    
    class Meta:
        verbose_name = "Категория справочника диагностики"
        verbose_name_plural = "Категории справочника диагностики"
        ordering = ['order', 'name']

class DiagnosticReferenceItem(models.Model):
    """
    Элементы справочника диагностики (конкретные признаки неисправностей)
    """
    category = models.ForeignKey(
        DiagnosticReferenceCategory, 
        on_delete=models.CASCADE, 
        related_name='items',
        verbose_name='Категория'
    )
    title = models.CharField('Заголовок', max_length=300)  # "Запотевание (тонкая масляная плёнка)"
    description = models.TextField('Описание', blank=True, null=True)
    
    # Где наблюдается
    observation_zones = models.TextField('Где наблюдается', blank=True, null=True)
    
    # Возможные причины
    possible_causes = models.TextField('Возможные причины', blank=True, null=True)
    
    # Последствия при игнорировании
    consequences = models.TextField('Последствия при игнорировании', blank=True, null=True)
    
    # Дополнительная информация
    additional_info = models.TextField('Дополнительная информация', blank=True, null=True)
    
    # Метаданные
    order = models.IntegerField('Порядок сортировки', default=0)
    is_active = models.BooleanField('Активна', default=True)
    created_at = models.DateTimeField('Дата создания', auto_now_add=True)
    updated_at = models.DateTimeField('Дата обновления', auto_now=True)
    
    class Meta:
        verbose_name = "Элемент справочника диагностики"
        verbose_name_plural = "Элементы справочника диагностики"
        ordering = ['category', 'order', 'title']
        indexes = [
            models.Index(fields=['category', 'is_active']),
        ]
```

## Этап 3: Backend (Views и API)

### 3.1 API для получения справочника

**Файл:** `portal/apps/register/views.py` или создать `portal/apps/register/api_views.py`

```python
from django.http import JsonResponse
from django.views.decorators.http import require_http_methods
from .models import DiagnosticReferenceCategory, DiagnosticReferenceItem

@require_http_methods(["GET"])
def diagnostic_reference_api(request):
    """
    API для получения справочника диагностики по категории
    GET /api/diagnostic_reference/?category=engine
    """
    category_code = request.GET.get('category', '')
    
    if not category_code:
        return JsonResponse({'error': 'Category parameter required'}, status=400)
    
    try:
        category = DiagnosticReferenceCategory.objects.get(code=category_code, is_active=True)
        items = DiagnosticReferenceItem.objects.filter(
            category=category,
            is_active=True
        ).order_by('order', 'title')
        
        data = {
            'category': {
                'code': category.code,
                'name': category.name,
                'description': category.description,
            },
            'items': [
                {
                    'id': item.id,
                    'title': item.title,
                    'description': item.description,
                    'observation_zones': item.observation_zones,
                    'possible_causes': item.possible_causes,
                    'consequences': item.consequences,
                    'additional_info': item.additional_info,
                }
                for item in items
            ]
        }
        return JsonResponse(data, safe=False)
    except DiagnosticReferenceCategory.DoesNotExist:
        return JsonResponse({'error': 'Category not found'}, status=404)
```

**JavaScript для работы со справочником:**

```javascript
// Загрузка справочника при открытии модального окна
$('#diagnosticReferenceModal').on('show.bs.modal', function (event) {
  const button = $(event.relatedTarget);
  const category = button.data('category');
  
  // Загрузка данных через AJAX
  $.ajax({
    url: '/api/diagnostic_reference/',
    data: { category: category },
    success: function(data) {
      $('#referenceCategoryName').text(data.category.name);
      renderReferenceItems(data.items);
    },
    error: function() {
      $('#referenceItemsList').html('<div class="alert alert-danger">Ошибка загрузки справочника</div>');
    }
  });
});

// Рендеринг элементов справочника
function renderReferenceItems(items) {
  const container = $('#referenceItemsList');
  container.empty();
  
  items.forEach(function(item) {
    const itemHtml = `
      <div class="card mb-3 reference-item" data-item-id="${item.id}">
        <div class="card-header d-flex justify-content-between align-items-center">
          <h6 class="mb-0">
            <i class="bi bi-file-text"></i> ${item.title}
          </h6>
          <div class="form-check">
            <input class="form-check-input reference-item-checkbox" type="checkbox" value="${item.id}" id="ref_${item.id}">
            <label class="form-check-label" for="ref_${item.id}">Выбрать</label>
          </div>
        </div>
        <div class="card-body">
          ${item.description ? `<p class="text-muted">${item.description}</p>` : ''}
          
          ${item.observation_zones ? `
            <div class="mb-2">
              <strong><i class="bi bi-geo-alt"></i> Где наблюдается:</strong>
              <div class="ms-3">${formatText(item.observation_zones)}</div>
            </div>
          ` : ''}
          
          ${item.possible_causes ? `
            <div class="mb-2">
              <strong><i class="bi bi-exclamation-triangle"></i> Возможные причины:</strong>
              <div class="ms-3">${formatText(item.possible_causes)}</div>
            </div>
          ` : ''}
          
          ${item.consequences ? `
            <div class="mb-2">
              <strong><i class="bi bi-x-circle text-danger"></i> Последствия при игнорировании:</strong>
              <div class="ms-3 text-danger">${formatText(item.consequences)}</div>
            </div>
          ` : ''}
          
          ${item.additional_info ? `
            <div class="alert alert-info mt-2 mb-0">
              <strong>Важно:</strong> ${item.additional_info}
            </div>
          ` : ''}
        </div>
      </div>
    `;
    container.append(itemHtml);
  });
}

// Форматирование текста (замена переносов строк на HTML)
function formatText(text) {
  return text.replace(/\n/g, '<br>').replace(/^[\s•\-]/gm, '• ');
}

// Поиск по справочнику
$('#referenceSearch').on('input', function() {
  const searchTerm = $(this).val().toLowerCase();
  $('.reference-item').each(function() {
    const text = $(this).text().toLowerCase();
    $(this).toggle(text.includes(searchTerm));
  });
});

// Применение выбранных элементов
$('#selectReferenceItems').on('click', function() {
  const selectedIds = $('.reference-item-checkbox:checked').map(function() {
    return $(this).val();
  }).get();
  
  // Сохранение выбранных элементов (можно добавить в скрытое поле формы)
  $('#selected_reference_items').val(selectedIds.join(','));
  
  // Обновление комментария (опционально - добавить названия выбранных элементов)
  // ...
  
  $('#diagnosticReferenceModal').modal('hide');
});
```
